# Dependencies for the LIGHTWEIGHT CLOUD RUNNER ONLY (cloud_runtime/app.py).
# Deliberately a small subset of the full requirements.txt (which also
# covers the desktop GUI, PDF/Word/Excel report generation, backtesting,
# and the local-only strategy-extraction pipeline -- none of that runs in
# the cloud). Keeping this separate keeps Railway's build fast and its
# image small; requirements.txt itself is untouched, so local development
# is completely unaffected by this file's existence.
#
# Traced by actually importing cloud_runtime.app's full dependency graph
# and listing every third-party package really touched -- not copied from
# requirements.txt by guesswork.
#
# 2026-09-16 audit: PINNED. Every package used to be unpinned, so each Render
# rebuild silently pulled the newest release of everything -- one breaking
# upstream release (pandas/fastapi/starlette/ccxt all ship often) would crash
# the next deploy with nobody around to notice. These are exactly the versions
# a clean `pip install -r` of the old unpinned file resolved on 2026-09-16
# (the same day the live service last rebuilt successfully -- same or near-
# identical versions), verified in a fresh venv: cloud_runtime.app imports,
# 223 routes register, /health returns ok, pip check clean; PyPI confirms
# manylinux cp313 wheels for every compiled pin (pandas, numpy, psycopg2-
# binary, psutil).
# To upgrade deliberately: bump a version here, redeploy, and check /health.
# starlette/pydantic/numpy are transitive but pinned too -- they are the
# ones most likely to break fastapi/pandas underneath an unchanged pin.

fastapi==0.141.1
starlette==1.6.0
pydantic==2.13.5
uvicorn[standard]==0.53.0
pandas==3.0.5
numpy==2.5.3
ccxt==4.5.78
